Load computed manifolds (run on PowerShell):

# Local destination
$destination = "C:\Users\mirei\OneDrive - epfl.ch\MA3\SdS\local\num_results_loc\computed_manifolds"

# Create destination folder if necessary
New-Item -ItemType Directory -Force -Path $destination | Out-Null

# Configuration IDs
$configs = @(11, 12, 13, 14, 15, 16, 17, 18, 21, 22, 23)

# Copy and rename each file
foreach ($id in $configs) {
    Write-Host "Downloading configuration $id..."

    scp "raventos@jed.hpc.epfl.ch:/home/raventos/SdS/cluster/manifold_results/config_$id/manifolds.npz" "$destination\manifold_$id.npz"
}
